        var exp = Expressionable.Create<TaktBomMaterialCost>();
        exp = exp.And(x => x.TenantCode == CurrentTenantCode && x.CompanyCode == CurrentCompanyCode);
        exp = exp.And(x => x.BomMaterialCostStatus == status);
