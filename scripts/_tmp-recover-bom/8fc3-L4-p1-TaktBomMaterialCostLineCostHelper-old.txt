using Takt.Domain.Entities.Logistics.Manufacturing.Bom;
using Takt.Shared.Helpers;

namespace Takt.Application.Services.Logistics.Manufacturing.Bom;

/// <summary>
/// BOM 物料成本行金额计算（无状态纯函数）
/// </summary>
public static class TaktBomMaterialCostLineCostHelper
{
    /// <summary>
    /// 将处理日期映射为期间键 yyyy-MM
    /// </summary>
    /// <param name="processedDate">处理日期</param>
    /// <returns>期间键</returns>
    public static string ToPeriodKey(DateTime processedDate)
    {
        return processedDate.ToString("yyyy-MM");
    }

    /// <summary>
    /// 是否参与 BOM 材料成本汇总（仅外部采购 F）
    /// </summary>
    /// <param name="row">成本行</param>
    /// <returns>是否计入产品 BOM 材料成本</returns>
    public static bool CountsTowardBomMaterialCost(TaktBomMaterialCost row)
    {
        ArgumentNullException.ThrowIfNull(row);
        return string.Equals(row.PurchaseType, "F", StringComparison.OrdinalIgnoreCase);
    }

    /// <summary>
    /// 筛选参与 BOM 材料成本汇总的行（PurchaseType=F）
    /// </summary>
    /// <param name="rows">成本行</param>
    /// <returns>外部采购行</returns>
    public static IEnumerable<TaktBomMaterialCost> FilterBomMaterialCostRows(IEnumerable<TaktBomMaterialCost> rows)
    {
        ArgumentNullException.ThrowIfNull(rows);
        return rows.Where(CountsTowardBomMaterialCost);
    }

    /// <summary>
    /// 计算单行组件成本（仅 F 行：移动平均价 ÷ 移动价格单位 × 组件用量；E 行返回 0）
    /// </summary>
    /// <param name="row">成本行</param>
    /// <returns>行成本</returns>
    public static decimal CalculateLineCost(TaktBomMaterialCost row)
    {
        ArgumentNullException.ThrowIfNull(row);
        if (!CountsTowardBomMaterialCost(row))
        {
            return 0m;
        }
        var priceUnit = ResolveMovingPriceUnit(row);
        return row.ComponentQuantity * row.MovingAveragePrice / priceUnit;
    }

    /// <summary>
    /// 取有效单价（汇总口径仅 F：移动平均价；E 返回 0）
    /// </summary>
    /// <param name="row">成本行</param>
    /// <returns>单价</returns>
    public static decimal ResolveEffectiveUnitPrice(TaktBomMaterialCost row)
    {
        ArgumentNullException.ThrowIfNull(row);
        if (!CountsTowardBomMaterialCost(row))
        {
            return 0m;
        }
        return row.MovingAveragePrice;
    }

    /// <summary>
    /// 取价格单位（汇总口径仅 F：移动价格单位，与移动平均价配对）
    /// </summary>
    /// <param name="row">成本行</param>
    /// <returns>价格单位</returns>
    public static int ResolvePriceUnit(TaktBomMaterialCost row)
    {
        ArgumentNullException.ThrowIfNull(row);
        if (!CountsTowardBomMaterialCost(row))
        {
            return 1;
        }
        return ResolveMovingPriceUnit(row);
    }

    /// <summary>
    /// 取移动价格单位（≤0 时按 1 处理）
    /// </summary>
    /// <param name="row">成本行</param>
    /// <returns>移动价格单位</returns>
    public static int ResolveMovingPriceUnit(TaktBomMaterialCost row)
    {
        ArgumentNullException.ThrowIfNull(row);
        return row.MovingPriceUnit <= 0 ? 1 : row.MovingPriceUnit;
    }

    /// <summary>
    /// 取货币码（汇总口径仅 F：移动价格货币）
    /// </summary>
    /// <param name="row">成本行</param>
    /// <returns>货币码</returns>
    public static string ResolveCurrency(TaktBomMaterialCost row)
    {
        ArgumentNullException.ThrowIfNull(row);
        if (!CountsTowardBomMaterialCost(row))
        {
            return string.Empty;
        }
        return row.MovingPriceCurrency ?? string.Empty;
    }

    /// <summary>
    /// 组件行唯一键（序号 + BOM 项目号 + 组件编码，与行唯一索引一致）
    /// </summary>
    /// <param name="row">成本行</param>
    /// <returns>组件键</returns>
    public static string BuildComponentKey(TaktBomMaterialCost row)
    {
        ArgumentNullException.ThrowIfNull(row);
        return $"{row.SequenceNo}|{row.BomItemNo}|{row.ComponentCode}";
    }

    /// <summary>
    /// 取产品在指定工厂、指定期间内各组件最新处理日期的外部采购（F）快照行（用于汇总与价格对比）
    /// </summary>
    /// <param name="rows">候选行</param>
    /// <param name="plantCode">工厂代码</param>
    /// <param name="productCode">产品编码</param>
    /// <param name="periodKey">期间键 yyyy-MM</param>
    /// <returns>快照行列表</returns>
    public static List<TaktBomMaterialCost> ResolvePeriodSnapshot(
        IEnumerable<TaktBomMaterialCost> rows,
        string plantCode,
        string productCode,
        string periodKey)
    {
        ArgumentNullException.ThrowIfNull(rows);
        ArgumentException.ThrowIfNullOrWhiteSpace(plantCode);
        ArgumentException.ThrowIfNullOrWhiteSpace(productCode);
        ArgumentException.ThrowIfNullOrWhiteSpace(periodKey);
        var periodRows = rows
            .Where(r => string.Equals(r.PlantCode, plantCode, StringComparison.OrdinalIgnoreCase)
                && ProductCodeMatches(r.ProductCode, productCode)
                && TaktBomMaterialCostLineCostHelper.ToPeriodKey(r.ProcessedDate) == periodKey)
            .ToList();
        if (periodRows.Count == 0)
        {
            return new List<TaktBomMaterialCost>();
        }
        return periodRows
            .Where(CountsTowardBomMaterialCost)
            .GroupBy(BuildComponentKey, StringComparer.Ordinal)
            .Select(g => g
                .OrderByDescending(r => r.ProcessedDate)
                .ThenByDescending(r => r.Id)
                .First())
            .ToList();
    }

    /// <summary>
    /// 汇总产品 BOM 材料成本（仅 F 行求和）
    /// </summary>
    /// <param name="snapshotRows">快照行</param>
    /// <returns>总成本</returns>
    public static decimal SumSnapshotCost(IEnumerable<TaktBomMaterialCost> snapshotRows)
    {
        ArgumentNullException.ThrowIfNull(snapshotRows);
        return FilterBomMaterialCostRows(snapshotRows).Sum(CalculateLineCost);
    }
