    /// <summary>
    /// 提交后台重算机种月平均材料成本（完成后 SignalR 通知触发用户；不含机种编码回填）
    /// </summary>
    /// <param name="queryDto">与列表相同的筛选条件（忽略分页）</param>
    /// <param name="forceRecalculate">为 true 时先清零再重算</param>
    /// <returns>已提交响应</returns>
    [TaktPermission("logistics:manufacturing:bom:material:cost:update", "重算BOM机种月平均材料成本")]
    [HttpPut("recalculate-model-average")]
    public async Task<IActionResult> RecalculateBomMaterialCostModelMonthlyAverageAsync(
        [FromQuery] TaktBomMaterialCostQueryDto queryDto,
        [FromQuery] bool forceRecalculate = false)
    {
        try
        {
            var prepared = TaktBomMaterialCostService.PrepareRecalculateModelAverageQuery(queryDto);
            await _recalculateBackgroundService.EnqueueRecalculateAsync(prepared.Query, forceRecalculate);
            var submitted = new TaktBomMaterialCostRecalculateSubmittedDto
            {
                ProcessedMonth = prepared.ProcessedMonth,
                ForceRecalculate = forceRecalculate,
            };
            return Success(submitted, "已提交后台重算，完成后将通知您");
        }
        catch (Exception ex)
        {
            return HandleException(ex);
        }
    }
